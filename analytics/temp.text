async function chart() {
    const apiUrl = '/analytics/mrbeast_data.json';
    const response = await fetch(apiUrl);
    const data = await response.json();

    // Ensure the date is a Date object
    data.items.forEach(item => {
        item.date = new Date(item.date);
    });

    // Calculate daily changes in subscribers and views
    const dailySubsChange = [];
    const dailyViewsChange = [];

    for (let i = 1; i < data.items.length; i++) {
        const prev = data.items[i - 1];
        const curr = data.items[i];
        const subsChange = curr.subscribers - prev.subscribers;
        const viewsChange = curr.views - prev.views;
        
        dailySubsChange.push([curr.date.getTime(), subsChange]);
        dailyViewsChange.push([curr.date.getTime(), viewsChange]);
    }

    const weeklySubsChange = [];
    const weeklyViewsChange = [];
    const monthlySubsChange = [];
    const monthlyViewsChange = [];

    const weeklyTotalSubs = [];
    const weeklyTotalViews = [];
    const monthlyTotalSubs = [];
    const monthlyTotalViews = [];

    // Helper function to get the start of the week (Sunday)
    function getWeekStart(date) {
        const d = new Date(date);
        const day = d.getDay();
        const diff = d.getDate() - day;
        return new Date(d.setDate(diff));
    }

    // Helper function to get the start of the month
    function getMonthStart(date) {
        return new Date(date.getFullYear(), date.getMonth(), 1);
    }

    // Group data by week
    const weeklyData = data.items.reduce((acc, item) => {
        const weekStart = getWeekStart(item.date).getTime();
        if (!acc[weekStart]) acc[weekStart] = [];
        acc[weekStart].push(item);
        return acc;
    }, {});

    // Calculate weekly changes and totals
    Object.keys(weeklyData).sort().forEach((week, index, weeks) => {
        const currWeek = weeklyData[week];
        const totalSubs = currWeek[currWeek.length - 1].subscribers;
        const totalViews = currWeek[currWeek.length - 1].views;

        weeklyTotalSubs.push([parseInt(week), totalSubs]);
        weeklyTotalViews.push([parseInt(week), totalViews]);

        if (index === 0) return; // skip the first week since there's no previous week to compare to
        const prevWeek = weeklyData[weeks[index - 1]];
        const subsChange = currWeek[currWeek.length - 1].subscribers - prevWeek[prevWeek.length - 1].subscribers;
        const viewsChange = currWeek[currWeek.length - 1].views - prevWeek[prevWeek.length - 1].views;

        weeklySubsChange.push([parseInt(week), subsChange]);
        weeklyViewsChange.push([parseInt(week), viewsChange]);
    });

    // Group data by month
    const monthlyData = data.items.reduce((acc, item) => {
        const monthStart = getMonthStart(item.date).getTime();
        if (!acc[monthStart]) acc[monthStart] = [];
        acc[monthStart].push(item);
        return acc;
    }, {});

    // Calculate monthly changes and totals
    Object.keys(monthlyData).sort().forEach((month, index, months) => {
        const currMonth = monthlyData[month];
        const totalSubs = currMonth[currMonth.length - 1].subscribers;
        const totalViews = currMonth[currMonth.length - 1].views;

        monthlyTotalSubs.push([parseInt(month), totalSubs]);
        monthlyTotalViews.push([parseInt(month), totalViews]);

        if (index === 0) return; // skip the first month since there's no previous month to compare to

        monthlyViewsChange.push([parseInt(month), viewsChange]);
    });

                }

                    fontSize: '20px'
                }
            }

            shared: true,
            valueSuffix: ' Subscribers'

        credits: { enabled: false },

            itemStyle: {
                fontSize: '20px',
                color: '#fff'
            }
